\section{Theory: Grover $\&$ Counting}

\begin{frame}{1. Grover's Algorithm: The Mechanism}
    \textbf{Goal:} Find marked element(s) $x$ such that $f(x)=1$ in an unsorted database of size $N=2^n$. (ex : Sudoku, Traveling Salesman Problem)
    
    \vspace{0.3cm}

    The algorithm iterates the \textbf{Grover Operator} $\hat{G} = (2|\psi\rangle\langle\psi| - I)\hat{O}$, where $|\psi\rangle = H^{\otimes n}|0\rangle$:
    \begin{itemize}
        \item \textbf{Oracle ($\hat{O}$):} Marks the solution by flipping its phase:$\hat{O}|x\rangle = (-1)^{f(x)}|x\rangle $
        \item \textbf{Diffuser:} Performs an inversion about the mean amplitude, amplifying the probability of the marked state.
    \end{itemize}

    \begin{center}
        \includegraphics[width=0.60\textwidth]{../asset/circuit_grover.png}
    \end{center}

    % Citations
    \blfootnote{\tiny L. K. Grover, Phys. Rev. Lett. \textbf{79}, 325 (1997).}
    \blfootnote{\tiny M. A. Nielsen \& I. L. Chuang, \textit{Quantum Computation and Quantum Information} (2010).}
\end{frame}

\begin{frame}{1. Grover's Algorithm: Geometric Picture}

    \textbf{Key Idea:}
    Grover’s algorithm acts as a \textbf{rotation} in a 2D subspace,
    progressively amplifying the target state.
    
    \begin{columns}[c]
        \column{0.55\textwidth}
        \begin{itemize}
            \item The quantum state lives in the plane spanned by:
            \begin{itemize}
                \item $|\beta\rangle$: superposition of \textbf{solutions}
                \item $|\alpha\rangle$: superposition of \textbf{non-solutions}
            \end{itemize}
            \item Each Grover iteration rotates the state by an angle $\theta$
            \item After an optimal number of iterations, the state aligns with
            $|\beta\rangle$
            \item Optimal Number of Iterations: $r_{\text{opt}} \approx \frac{\pi}{4}\sqrt{\frac{N}{m}}$
        \end{itemize}
    
        \column{0.40\textwidth}
        \centering
        \includegraphics[width=\textwidth]{../asset/geom.png}
        
        \small
        \textit{Rotation in the subspace spanned by $|\alpha\rangle$ and $|\beta\rangle$.}
    \end{columns}
    
\end{frame}

% --- DÉPLACÉ ICI : La théorie du Counting AVANT les résultats ---
\begin{frame}{1. Quantum Counting Algorithm}
    \textbf{Problem:} Grover's optimal iterations $r$ depend on $m$. What if $m$ is unknown?
    
    \begin{block}{The Solution}
        Combine Grover operator $G$ with \textbf{Quantum Phase Estimation (QPE)}.
    \end{block}

    \begin{columns}[c] % [c] aligne verticalement le texte et l'image au centre
        
        % Colonne de gauche : Texte
        \begin{column}{0.45\textwidth}
            \begin{itemize}
                \item $G$ has eigenvalues $e^{\pm i\theta}$.
                \item \textbf{Phase Kickback} : transfer $\theta$ into the counting register.
                \item \textbf{Inverse QFT} : reveals the phase $\theta$.
                \item we measure $m$: $m \approx N \sin^2(\theta/2) $
            \end{itemize}
        \end{column}
        
        % Colonne de droite : Image
        \begin{column}{0.45\textwidth}
            \centering
            % width=\textwidth ici signifie "100% de la largeur de la colonne" (donc 55% de la page)
            \includegraphics[width=\textwidth]{../asset/circuit_quant_count.png}
        \end{column}
        
    \end{columns}
    \blfootnote{\tiny G. Brassard et al., \textit{Quantum Amplitude Amplification and Estimation}, arXiv:quant-ph/0005055 (2000).}
\end{frame}

\section{Complexity Analysis}

\begin{frame}{2. Computational Complexity : Limits of Speedup}
    \begin{columns}
        % --- Colonne Gauche : L'Optimalité (Théorie pure) ---
        \column{0.5\textwidth}
        \textbf{1. The Quadratic Speedup}
        \begin{itemize}
            \item Classical: $\mathcal{O}(N)$
            \item Grover: $\mathcal{O}(\sqrt{N})$
        \end{itemize}
        
        \vspace{0.4cm}
        
        \textbf{2. Is further speedup possible?}
        \begin{block}{BBBV Theorem (1997)}
            Proven lower bound: Any quantum black-box search requires $\Omega(\sqrt{N})$ queries.
        \end{block}
        
        \textbf{Consequence:} Grover is \textbf{asymptotically optimal}. We cannot do better than $\sqrt{N}$.

        % --- Colonne Droite : Diagramme de Complexité (Visuel) ---
        \column{0.5\textwidth}
        \centering
        \textbf{3. Can it solve NP-Complete?}

\begin{frame}{A.1. Grover: Theory and Optimal Performance}

    \footnotesize
    
    \textbf{Subspace Definitions:}
    For $m$ solutions in a database of size $N = 2^n$:
    \begin{align*}
        |\beta\rangle &= \frac{1}{\sqrt{m}} \sum_{x \in S} |x\rangle \qquad
        |\alpha\rangle = \frac{1}{\sqrt{N-m}} \sum_{x \notin S} |x\rangle
    \end{align*}
    
    \textbf{Initial State Decomposition:}
    $|\psi\rangle = \cos(\theta/2)|\alpha\rangle + \sin(\theta/2)|\beta\rangle,
    \quad
    \theta = 2\arcsin\!\left(\sqrt{\frac{m}{N}}\right)$
    
    \textbf{Success Probability:} $P(r) = \sin^2\!\left((2r+1)\frac{\theta}{2}\right)$
    \textbf{Optimal Number of Iterations:}$r_{\text{opt}} \approx \frac{\pi}{4}\sqrt{\frac{N}{m}}$
    
    \begin{columns}[c]
        \column{0.55\textwidth}
        \textbf{Example:} $n=3$, $m=1$
        \begin{itemize}
            \item $r_{\text{opt}} \approx 2$
            \item $P_{\text{success}} \approx 0.945$
        \end{itemize}
    
        \column{0.40\textwidth}
        \centering
        \includegraphics[width=\textwidth]{../asset/grover_simulated_probability_n3_m1.png}
        
        \tiny
        \textit{Theoretical success probability vs Grover iterations.}
    \end{columns}
    
    \end{frame}

\begin{frame}{A.2. Quantum Phase Estimation (QPE) : Ideal vs Noisy Simulation}
    \begin{columns}[c]
        % --- Colonne Gauche : Paramètres & Analyse ---
        \column{0.45\textwidth}
            \begin{block}{Simulation Setup}
                \footnotesize
                \begin{itemize}
                    \item \textbf{System:} $n=3$, $m=1$ solutions.
                    \item \textbf{Precision:} $t=6$ qubits.
                    \item \textbf{Noise:} Depolarizing ($\epsilon = 10^{-5}$).
                \end{itemize}
            \end{block}
            
            \vspace{0.3cm}
            
            \textbf{Spectral Interpretation:}
            \footnotesize
            The Grover operator $G$ acts as a rotation in a 2D subspace, possessing conjugate eigenvalues $e^{\pm i\theta}$.
            QPE projects the state onto both eigenphases:
            \begin{itemize}
                \item Peak 1: $\phi_1 \approx \theta \rightarrow$ integer $j$
                \item Peak 2: $\phi_2 \approx 2\pi - \theta \rightarrow$ integer $2^t - j$
            \end{itemize}
            
            \vspace{0.2cm}
            \textbf{Solution Invariance:}
            Due to the symmetry $\sin^2(x) = \sin^2(\pi-x)$, both peaks yield the exact same count:
            $$ m = N \sin^2\left(\frac{\theta}{2}\right) = N \sin^2\left(\frac{2\pi - \theta}{2}\right) $$

        % --- Colonne Droite : Graphique ---
        \column{0.55\textwidth}
            \centering
            \includegraphics[width=\textwidth]{../asset/quantum_counting_ideal_vs_noisy_n3_t6.png}
            
            \vspace{0.1cm}
            \scriptsize \textit{Fig: Number of targets ($m$). The two symmetric peaks physically correspond to the forward and backward rotation of the Grover operator. ($n = 3, j = \{6, 58\}, \text{shots }=  2\;000$, AerSimulator())}
    \end{columns}
\end{frame}

\begin{frame}{A.3. Grover vs. Shor Strategy}

                \item \textbf{Impact:} Solves Factorization efficiently (In BQP).

\begin{frame}{A.4. From Grover to Digital Quantum Simulation}
    
    \begin{columns}[t]
        % --- Colonne 1 : Ton Projet (Quantum Counting) ---
        \column{0.45\textwidth}
        \begin{block}{1. My Project (Quantum Counting)}
            \textbf{Goal:} Find the number of solutions $m$.
            \vspace{0.2cm}
            \begin{itemize}
                \item \textbf{Operator:} Grover Iterate $G$.

\begin{frame}{1. Grover's Algorithm: Geometric Interpretation}

    \textbf{Geometric Interpretation:}
    Rotates the state vector by angle $\theta$ towards the target state $|\beta\rangle$.
    
    \begin{columns}[c]
        % --- Colonne Gauche : Mathématiques Rigoureuses ---
        \begin{column}{0.6\textwidth}
            \footnotesize % On réduit la police pour faire tenir les maths
            
            \textbf{1. Subspace Definitions:}
            Let $S$ be the set of $m$ solutions. We define the normalized basis states:
            \begin{align*}
                |\beta\rangle &= \frac{1}{\sqrt{m}} \sum_{x \in S} |x\rangle \quad \text{(Target superpos.)} \\
                |\alpha\rangle &= \frac{1}{\sqrt{N-m}} \sum_{x \notin S} |x\rangle \quad \text{(Non-target)}
            \end{align*}
    
            \textbf{2. Initial State Decomposition:}
            The initial state $|\psi\rangle = H^{\otimes n}|0\rangle$ lies in the plane spanned by $|\alpha\rangle$ and $|\beta\rangle$:
            $$ |\psi\rangle = \sqrt{\frac{N-m}{N}}|\alpha\rangle + \sqrt{\frac{m}{N}}|\beta\rangle $$
            $$ |\psi\rangle = \cos(\theta/2)|\alpha\rangle + \sin(\theta/2)|\beta\rangle $$
            
            \textbf{3. Geometric Action:}
            The Grover operator $G$ performs a rotation of angle $\theta$ in this 2D plane:
            $$ \theta = 2\arcsin(\sqrt{m/N}) \approx 2\sqrt{m/N} $$
        \end{column}
